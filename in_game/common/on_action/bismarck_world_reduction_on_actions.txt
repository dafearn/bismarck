# Vanilla 1.4 fires this once on a new game, after players have picked their countries.
on_game_started_after_lobby = {
	on_actions = {
		bismarck_wr_on_game_started
	}
}

bismarck_wr_on_game_started = {
	effect = {
		bismarck_wr_apply = yes
	}
}

# The Great Pestilence (the Americas epidemic) only ends after the Andes have been infected.
# With South America removed it can get stuck, so end it once the disease itself has burned out.
yearly_country_pulse = {
	on_actions = {
		bismarck_wr_end_stuck_pestilence
	}
}

bismarck_wr_end_stuck_pestilence = {
	trigger = {
		is_ai = no
		has_global_variable = bismarck_wr_south_america_removed
		is_situation_active = situation:great_pestilence
		NOT = { disease_is_active = disease:great_pestilence }
	}
	effect = {
		end_situation = situation:great_pestilence
	}
}
